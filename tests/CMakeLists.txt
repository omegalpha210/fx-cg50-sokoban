cmake_minimum_required(VERSION 3.15)
project(SokobanHost C)
set(CMAKE_C_STANDARD 11)
option(SOK_SANITIZE "Enable undefined behavior sanitizer" OFF)
add_compile_options(-Wall -Wextra -Werror -pedantic -Wshadow -Wconversion -g -UNDEBUG)
if(SOK_SANITIZE)
  add_compile_options(-fsanitize=undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  add_link_options(-fsanitize=undefined -fno-sanitize-recover=all)
endif()
option(SOK_ASAN "Also enable AddressSanitizer (requires working host runtime)" OFF)
if(SOK_ASAN)
  add_compile_options(-fsanitize=address)
  add_link_options(-fsanitize=address)
endif()
include_directories(../include)
add_library(core ../src/game/game.c ../src/maps/generated_maps.c
  ../src/storage/progress.c ../src/storage/codec.c ../src/storage/transaction.c
  ../src/app.c ../src/ui/input.c)
enable_testing()
foreach(name engine storage workflow)
  add_executable(test_${name} test_${name}.c)
  target_link_libraries(test_${name} core)
  add_test(NAME ${name} COMMAND test_${name})
endforeach()
add_executable(capture capture.c host/display.c ../src/ui/render.c)
target_include_directories(capture PRIVATE host)
target_link_libraries(capture core)
add_test(NAME renderer COMMAND capture ${CMAKE_CURRENT_BINARY_DIR}/captures)
add_executable(test_storage_native test_storage_native.c ../src/storage/native.c)
target_include_directories(test_storage_native PRIVATE native-storage)
target_compile_definitions(test_storage_native PRIVATE FXCG50)
target_link_libraries(test_storage_native core)
add_test(NAME storage_native COMMAND test_storage_native)
add_executable(test_power test_power.c)
target_include_directories(test_power PRIVATE native-input)
target_link_libraries(test_power core)
add_test(NAME power COMMAND test_power)
# Public illustrations never link the upstream map pack.
add_executable(capture_public public/capture.c public/fixture.c host/display.c
  ../src/game/game.c ../src/ui/render.c)
target_include_directories(capture_public PRIVATE host)
add_test(NAME renderer_public COMMAND capture_public ${CMAKE_CURRENT_BINARY_DIR}/public-captures)
